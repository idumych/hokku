# Options for enabling/disabling backends
option('enable_wayland_backend',
  type : 'boolean',
  value : true,
  description : 'Enable Wayland backend')

option('enable_x11_backend',
  type : 'boolean',
  value : true,
  description : 'Enable X11 backend')

option('enable_unit_test_compile',
  type : 'boolean',
  value : true,
  description : 'Enable compilation of unit tests')

option('enable_apps',
  type : 'boolean',
  value : true,
  description : 'Build the bundled apps and preferences (off: libraries, add-ons and tools only, e.g. for bundling with one app)')

option('enable_libgame',
  type : 'boolean',
  value : true,
  description : 'Build Game Kit as standalone libgame')

option('labwc_hokku',
  type : 'feature',
  value : 'auto',
  description : 'Build labwc-hokku, labwc drawing other apps\' windows in Hokku\'s frames, for the desktop session (fetches labwc, and wlroots if the system\'s is older than 0.20)')
